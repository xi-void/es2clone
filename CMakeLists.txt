cmake_minimum_required(VERSION 3.16)
project(earthsiege_resistance VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile>=1.1)
add_library(simulation src/simulation.cpp src/locomotion.cpp)
target_include_directories(simulation PUBLIC src)
target_compile_options(simulation PRIVATE -Wall -Wextra -Wpedantic)
add_library(mech_audio src/audio.cpp src/synthesis.cpp src/audio_assets.cpp src/music_stream.cpp)
target_link_libraries(mech_audio PUBLIC simulation PkgConfig::SDL2 PkgConfig::SNDFILE Threads::Threads)
target_compile_options(mech_audio PRIVATE -Wall -Wextra -Wpedantic)
add_library(original_graphics src/original_graphics.cpp)
target_include_directories(original_graphics PUBLIC src)
target_compile_options(original_graphics PRIVATE -Wall -Wextra -Wpedantic)
add_executable(earthsiege src/main.cpp src/renderer.cpp src/original_cockpit.cpp)
target_link_libraries(earthsiege PRIVATE mech_audio original_graphics PkgConfig::SDL2_TTF OpenGL::GL)
target_compile_options(earthsiege PRIVATE -Wall -Wextra -Wpedantic)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(TARGET earthsiege POST_BUILD COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/stage_public_assets.py ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:earthsiege>/assets)
enable_testing()
add_executable(original_graphics_tests tests/original_graphics_tests.cpp)
target_link_libraries(original_graphics_tests PRIVATE original_graphics)
add_test(NAME original_graphics COMMAND original_graphics_tests)
add_executable(graphics_probe tools/graphics_probe.cpp)
target_link_libraries(graphics_probe PRIVATE original_graphics)
add_test(NAME graphics_importer COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/import_graphics_tests.py)
add_test(NAME audio_importer COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/import_audio_tests.py)
add_executable(simulation_tests tests/simulation_tests.cpp)
target_link_libraries(simulation_tests PRIVATE simulation)
add_test(NAME simulation COMMAND simulation_tests)
add_executable(locomotion_tests tests/locomotion_tests.cpp)
target_link_libraries(locomotion_tests PRIVATE simulation)
add_test(NAME locomotion COMMAND locomotion_tests)
add_executable(audio_tests tests/audio_tests.cpp)
target_link_libraries(audio_tests PRIVATE mech_audio)
add_test(NAME audio COMMAND audio_tests)
set_tests_properties(audio PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy" TIMEOUT 60)
add_executable(audio_assets_tests tests/audio_assets_tests.cpp)
target_link_libraries(audio_assets_tests PRIVATE mech_audio)
add_test(NAME audio_assets COMMAND audio_assets_tests)
set_tests_properties(audio_assets PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy" TIMEOUT 60)
add_executable(audio_preview tools/audio_preview.cpp)
target_link_libraries(audio_preview PRIVATE mech_audio)
add_executable(battlefield_render tests/battlefield_render.cpp src/renderer.cpp src/original_cockpit.cpp)
target_link_libraries(battlefield_render PRIVATE mech_audio original_graphics PkgConfig::SDL2_TTF OpenGL::GL)
install(TARGETS earthsiege RUNTIME DESTINATION bin)
install(DIRECTORY assets/ DESTINATION share/earthsiege/assets
    FILES_MATCHING PATTERN "*.ttf" PATTERN "*.svg" PATTERN "*.desktop"
    PATTERN "*.md" PATTERN "*.txt" PATTERN "LICENSE*" PATTERN "audio.ini"
    PATTERN "original" EXCLUDE)
install(FILES assets/earthsiege.desktop DESTINATION share/applications)
install(FILES assets/earthsiege.svg DESTINATION share/icons/hicolor/scalable/apps)
